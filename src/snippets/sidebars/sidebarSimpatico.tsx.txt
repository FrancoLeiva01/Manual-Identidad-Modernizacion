import { useState } from "react";
import {
  CalendarDays,
  ChevronDown,
  ClipboardList,
  HelpCircle,
  LayoutDashboard,
  Settings,
  Users,
} from "lucide-react";

const grupos = [
  {
    titulo: "Asistencias",
    icono: ClipboardList,
    items: [
      { label: "Listado de agentes", badge: 0 },
      { label: "Cargar horarios", badge: 0 },
      { label: "Pendientes", badge: 12 },
    ],
  },
  {
    titulo: "Personal",
    icono: Users,
    items: [
      { label: "Legajos", badge: 0 },
      { label: "Licencias", badge: 3 },
    ],
  },
  {
    titulo: "Parámetros",
    icono: Settings,
    items: [
      { label: "Feriados", badge: 0 },
      { label: "Horarios", badge: 0 },
    ],
  },
];

// Sidebar de alto completo, flotante, con grupos desplegables
export default function SidebarSimpatico() {
  const [abiertos, setAbiertos] = useState<string[]>(["Asistencias"]);
  const [activo, setActivo] = useState("Dashboard");

  const toggle = (t: string) =>
    setAbiertos((a) => (a.includes(t) ? a.filter((x) => x !== t) : [...a, t]));

  const claseItem = (label: string) =>
    "flex w-full items-center justify-between rounded-xl px-3 py-2 text-left text-sm transition " +
    (activo === label
      ? "bg-sky-100 font-semibold text-sky-800"
      : "text-slate-600 hover:bg-slate-100 hover:text-slate-900");

  return (
    <div className="sticky top-0 h-screen shrink-0 p-3">
      <aside className="flex h-full w-64 flex-col rounded-3xl bg-white p-4 shadow-xl ring-1 ring-slate-200">
        {/* Marca */}
        <div className="mb-6 flex items-center gap-3 px-2">
          <div className="flex h-10 w-10 items-center justify-center rounded-2xl bg-sky-600 font-black text-white">
            RH
          </div>
          <div>
            <p className="font-bold leading-tight text-slate-900">Portal RRHH</p>
            <p className="text-xs text-slate-500">Asistencia</p>
          </div>
        </div>

        <nav className="flex-1 space-y-1 overflow-y-auto">
          <button
            type="button"
            onClick={() => setActivo("Dashboard")}
            className={claseItem("Dashboard")}
          >
            <span className="flex items-center gap-3">
              <LayoutDashboard size={18} />
              Dashboard
            </span>
          </button>

          <p className="px-3 pt-4 pb-1 text-[11px] font-bold uppercase tracking-widest text-slate-400">
            Principal
          </p>

          {grupos.map(({ titulo, icono: Icono, items }) => {
            const abierto = abiertos.includes(titulo);
            return (
              <div key={titulo}>
                <button
                  type="button"
                  onClick={() => toggle(titulo)}
                  className="flex w-full items-center justify-between rounded-xl px-3 py-2 text-sm font-medium text-slate-700 transition hover:bg-slate-100"
                >
                  <span className="flex items-center gap-3">
                    <Icono size={18} className="text-slate-500" />
                    {titulo}
                  </span>
                  <ChevronDown
                    size={16}
                    className={"text-slate-400 transition-transform " + (abierto ? "rotate-180" : "")}
                  />
                </button>

                {/* Animación de apertura con grid-rows */}
                <div
                  className={
                    "grid transition-all duration-300 " +
                    (abierto ? "grid-rows-[1fr] opacity-100" : "grid-rows-[0fr] opacity-0")
                  }
                >
                  <div className="overflow-hidden">
                    <div className="ml-5 space-y-0.5 border-l-2 border-slate-100 py-1 pl-3">
                      {items.map(({ label, badge }) => (
                        <button
                          key={label}
                          type="button"
                          onClick={() => setActivo(label)}
                          className={claseItem(label)}
                        >
                          {label}
                          {badge > 0 && (
                            <span className="rounded-full bg-orange-500 px-2 py-0.5 text-[10px] font-bold text-white">
                              {badge}
                            </span>
                          )}
                        </button>
                      ))}
                    </div>
                  </div>
                </div>
              </div>
            );
          })}

          <button
            type="button"
            onClick={() => setActivo("Calendario")}
            className={claseItem("Calendario")}
          >
            <span className="flex items-center gap-3">
              <CalendarDays size={18} />
              Calendario
            </span>
          </button>
        </nav>

        {/* Tarjeta de ayuda */}
        <div className="mt-4 rounded-2xl bg-gradient-to-br from-sky-50 to-orange-50 p-4 ring-1 ring-sky-100">
          <HelpCircle size={22} className="mb-2 text-sky-600" />
          <p className="text-sm font-semibold text-slate-800">¿Te perdiste?</p>
          <p className="mt-0.5 text-xs text-slate-500">
            El manual de uso tiene todo explicado paso a paso.
          </p>
          <a
            href="#"
            className="mt-3 inline-block rounded-full bg-white px-3 py-1.5 text-xs font-semibold text-sky-700 shadow-sm transition hover:shadow"
          >
            Ver manual
          </a>
        </div>
      </aside>
    </div>
  );
}
