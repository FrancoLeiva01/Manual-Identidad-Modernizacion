import { useState } from "react";
import { Menu, X } from "lucide-react";

const links = ["Inicio", "Trámites", "Reportes", "Ayuda"];

export default function NavbarBasico() {
  const [abierto, setAbierto] = useState(false);

  return (
    <header className="border-b border-gray-200 bg-white">
      <div className="mx-auto flex h-16 max-w-7xl items-center justify-between px-6">
        <img
          src="/-Isologotipo_Catamarca Capital.png"
          alt="Catamarca Capital"
          className="h-10 w-auto"
        />

        {/* Escritorio */}
        <nav className="hidden items-center gap-6 text-sm text-gray-700 md:flex">
          {links.map((l) => (
            <a key={l} href="#" className="hover:text-sky-700">
              {l}
            </a>
          ))}
          <a
            href="#"
            className="rounded-md bg-sky-700 px-4 py-2 font-medium text-white hover:bg-sky-800"
          >
            Ingresar
          </a>
        </nav>

        {/* Botón menú celular */}
        <button
          type="button"
          onClick={() => setAbierto(!abierto)}
          aria-label="Abrir menú"
          className="rounded-md p-2 text-gray-700 hover:bg-gray-100 md:hidden"
        >
          {abierto ? <X size={22} /> : <Menu size={22} />}
        </button>
      </div>

      {/* Menú celular */}
      {abierto && (
        <nav className="flex flex-col border-t border-gray-200 px-6 py-3 text-sm text-gray-700 md:hidden">
          {links.map((l) => (
            <a key={l} href="#" className="py-2 hover:text-sky-700">
              {l}
            </a>
          ))}
        </nav>
      )}
    </header>
  );
}
