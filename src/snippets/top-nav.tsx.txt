import { useState } from 'react';
import { Link, useLocation } from 'react-router-dom';
import { ChevronRight } from 'lucide-react';
import { cn } from '@/lib/utils';

import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuTrigger,
} from '@/components/ui/dropdown-menu';

import { ModeToggle } from '@/components/mode-toggle';
import { Profile } from '@/app/layout/components';
import { NotificacionesBell } from './notifications-bell';
import { RevisionDependenciaModal } from './revision-dependencia-modal';

interface BreadcrumbItem {
  label: string;
  href?: string;
}

const adicionalesNombres: Record<string, string> = {
  afe: 'AFE',
  extras: 'Extras',
  patrimonial: 'Resp. Patrimonial',
  despacho: 'Despacho',
};

function buildBreadcrumbs(pathname: string): BreadcrumbItem[] {
  const items: BreadcrumbItem[] = [{ label: 'RRHH', href: '/dashboard' }];

  if (pathname === '/' || pathname === '/dashboard') {
    items.push({ label: 'Dashboard' });
    return items;
  }

  if (pathname.startsWith('/asistencia')) {
    items.push({ label: 'Asistencias' });

    if (pathname.startsWith('/asistencia/docentes/informes')) {
      items.push({ label: 'Docentes', href: '/asistencia/docentes' });
      items.push({ label: 'Informes' });
    } else if (pathname.startsWith('/asistencia/docentes')) {
      items.push({ label: 'Listado de Docentes' });
    } else if (pathname.startsWith('/asistencia/agentes')) {
      items.push({ label: 'Listado de Agentes' });
    } else if (pathname.startsWith('/asistencia/agente/')) {
      items.push({ label: 'Listado de Agentes', href: '/asistencia/agentes' });
      items.push({ label: 'Detalle de Agente' });
    } else if (pathname.startsWith('/asistencia/cargar/')) {
      items.push({ label: 'Listado de Agentes', href: '/asistencia/agentes' });
      items.push({ label: 'Cargar Asistencia' });
    } else if (pathname.startsWith('/asistencia/carga/previsulizar')) {
      items.push({ label: 'Listado de Agentes', href: '/asistencia/agentes' });
      items.push({ label: 'Previsualizar' });
    } else if (pathname.startsWith('/asistencia/reportes')) {
      items.push({ label: 'Reportes' });
    } else if (pathname.startsWith('/asistencia/settear-parametros')) {
      items.push({ label: 'Listado de Agentes', href: '/asistencia/agentes' });
      items.push({ label: 'Parámetros' });
    }

    return items;
  }

  if (pathname.startsWith('/cargar-horarios')) {
    items.push({ label: 'Asistencias' });
    items.push({ label: 'Cargar Horarios' });
    return items;
  }

  if (pathname.startsWith('/cargar-docentes')) {
    items.push({ label: 'Docentes' });
    items.push({ label: 'Cargar Horarios' });
    return items;
  }

  if (pathname.startsWith('/adicionales')) {
    items.push({ label: 'Adicionales' });
    const tipo = pathname.split('/')[2];
    if (tipo) {
      const nombre = adicionalesNombres[tipo] || tipo.toUpperCase();
      items.push({ label: nombre });
    }
    return items;
  }

  if (pathname.startsWith('/estados-asistencias')) {
    items.push({ label: 'Parámetros' });
    if (pathname.includes('/registro')) {
      items.push({ label: 'Estados', href: '/estados-asistencias' });
      items.push({ label: 'Nuevo Estado' });
    } else if (pathname.includes('/editar')) {
      items.push({ label: 'Estados', href: '/estados-asistencias' });
      items.push({ label: 'Editar Estado' });
    } else {
      items.push({ label: 'Estados' });
    }
    return items;
  }

  if (pathname.startsWith('/plantillas-turno')) {
    items.push({ label: 'Parámetros' });
    if (pathname.includes('/registro')) {
      items.push({ label: 'Horarios', href: '/plantillas-turno' });
      items.push({ label: 'Nuevo Horario' });
    } else if (pathname.includes('/editar')) {
      items.push({ label: 'Horarios', href: '/plantillas-turno' });
      items.push({ label: 'Editar Horario' });
    } else if (pathname.includes('/empleados-por-plantilla')) {
      items.push({ label: 'Horarios', href: '/plantillas-turno' });
      items.push({ label: 'Empleados por Plantilla' });
    } else {
      items.push({ label: 'Horarios' });
    }
    return items;
  }

  if (pathname.startsWith('/feriados')) {
    items.push({ label: 'Parámetros' });
    if (pathname.includes('/registro')) {
      items.push({ label: 'Feriados', href: '/feriados' });
      items.push({ label: 'Nuevo Feriado' });
    } else if (pathname.includes('/editar')) {
      items.push({ label: 'Feriados', href: '/feriados' });
      items.push({ label: 'Editar Feriado' });
    } else {
      items.push({ label: 'Feriados' });
    }
    return items;
  }

  if (pathname.startsWith('/logs')) {
    items.push({ label: 'Parámetros' });
    items.push({ label: 'Logs' });
    return items;
  }

  // Fallback genérico por partes de la URL
  const parts = pathname.split('/').filter(Boolean);
  parts.forEach((part, index) => {
    const isLast = index === parts.length - 1;
    const formatted = part.charAt(0).toUpperCase() + part.slice(1).replace(/-/g, ' ');
    items.push({
      label: formatted,
      href: isLast ? undefined : `/${parts.slice(0, index + 1).join('/')}`,
    });
  });

  return items;
}

export function TopNav() {
  const location = useLocation();
  const breadcrumbs = buildBreadcrumbs(location.pathname);
  const [dropdownOpen, setDropdownOpen] = useState(false);
  const [openRevisionArea, setOpenRevisionArea] = useState(false);

  return (
    <nav className="sticky top-0 z-40 p-4 sm:px-6 flex items-center justify-between bg-background h-full border-b border-border">
      <div className="font-medium text-sm flex items-center space-x-1 truncate max-w-full pl-10 xl:pl-0">
        {breadcrumbs.map((item, index) => {
          const isLast = index === breadcrumbs.length - 1;

          return (
            <div key={`${item.label}-${index}`} className="flex items-center">
              {index > 0 && (
                <ChevronRight className="h-3.5 w-3.5 text-muted-foreground/70 mx-1 shrink-0" />
              )}
              {item.href && !isLast ? (
                <Link
                  to={item.href}
                  className="text-muted-foreground hover:text-foreground transition-colors font-medium text-xs sm:text-sm"
                >
                  {item.label}
                </Link>
              ) : (
                <span
                  className={cn(
                    'text-xs sm:text-sm',
                    isLast ? 'text-primary font-semibold' : 'text-muted-foreground'
                  )}
                >
                  {item.label}
                </span>
              )}
            </div>
          );
        })}
      </div>

      <div className="flex items-center gap-4 sm:gap-3 ml-auto sm:ml-0">
        <NotificacionesBell />
        <ModeToggle />

        <DropdownMenu open={dropdownOpen} onOpenChange={setDropdownOpen} modal={false}>
          <DropdownMenuTrigger className="focus:outline-none ml-1">
            <img
              src="https://github.com/shadcn.png"
              alt="User avatar"
              width={28}
              height={28}
              className="rounded-full ring-2 ring-border sm:w-8 sm:h-8 cursor-pointer"
            />
          </DropdownMenuTrigger>
          <DropdownMenuContent
            align="end"
            sideOffset={8}
            className="w-70 sm:w-80 bg-background border-border rounded-lg shadow-lg"
          >
            <Profile
              avatar="https://github.com/shadcn.png"
              onOpenRevisionArea={() => {
                setDropdownOpen(false);
                setOpenRevisionArea(true);
              }}
            />
          </DropdownMenuContent>
        </DropdownMenu>

        {/* Modal de Revisión por Dependencia fuera del DropdownMenu para evitar trampas de foco */}
        <RevisionDependenciaModal
          open={openRevisionArea}
          onOpenChange={setOpenRevisionArea}
        />
      </div>
    </nav>
  );
}
