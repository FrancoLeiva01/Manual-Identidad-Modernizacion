// Línea con relleno degradé, en SVG puro (sin librerías)
function Sparkline({ datos }: { datos: number[] }) {
  const max = Math.max(...datos);
  const pts = datos.map((d, i) => [(i / (datos.length - 1)) * 100, 40 - (d / max) * 36]);
  const linea = pts.map(([x, y], i) => (i ? "L" : "M") + x + "," + y).join(" ");
  return (
    <svg viewBox="0 0 100 40" preserveAspectRatio="none" className="h-16 w-full">
      <defs>
        <linearGradient id="relleno" x1="0" x2="0" y1="0" y2="1">
          <stop offset="0%" stopColor="#f97316" stopOpacity="0.35" />
          <stop offset="100%" stopColor="#f97316" stopOpacity="0" />
        </linearGradient>
      </defs>
      <path d={linea + " L100,40 L0,40 Z"} fill="url(#relleno)" />
      <path d={linea} fill="none" stroke="#f97316" strokeWidth="1.5" vectorEffect="non-scaling-stroke" />
    </svg>
  );
}

// Anillo de progreso
function Anillo({ valor }: { valor: number }) {
  const r = 42;
  const c = 2 * Math.PI * r;
  return (
    <svg viewBox="0 0 100 100" className="h-32 w-32 -rotate-90">
      <circle cx="50" cy="50" r={r} fill="none" stroke="currentColor" strokeWidth="8" className="text-slate-100" />
      <circle
        cx="50"
        cy="50"
        r={r}
        fill="none"
        stroke="url(#anillo)"
        strokeWidth="8"
        strokeLinecap="round"
        strokeDasharray={c}
        strokeDashoffset={c * (1 - valor / 100)}
        className="transition-[stroke-dashoffset] duration-1000"
      />
      <defs>
        <linearGradient id="anillo">
          <stop offset="0%" stopColor="#0ea5e9" />
          <stop offset="100%" stopColor="#10b981" />
        </linearGradient>
      </defs>
    </svg>
  );
}
